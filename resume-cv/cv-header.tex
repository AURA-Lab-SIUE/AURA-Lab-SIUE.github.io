% LaTeX header for CV.md → PDF rendering via pandoc + xelatex
% Pairs with CV.md in this directory. Usage:
%   pandoc CV.md -o Leith_CV.pdf --pdf-engine=xelatex \
%     --include-in-header=cv-header.tex -V geometry:margin=0.75in \
%     -V fontsize=10.5pt -V lang=en-US
%
% Fonts are declared HERE via fontspec, not on the command line. The eight
% static faces in ./fonts/ are instanced from the Google Fonts variable
% originals by ../make_fonts.py. Both families are SIL OFL, same as on the site.
%
% Section hierarchy from the CV.md:
%   \subsection (## in markdown) — major sections: Academic Appointments, Publications, etc.
%   \subsubsection (### in markdown) — sub-buckets: Editorials, External Funding (Completed), etc.

% ── Identity ────────────────────────────────────────────────────────────────
% "Instrument" — the shared AURA Lab x SIM DAD design system from apleith.com,
% rendered in light for print. Display face Archivo, body face Jakarta (Plus
% Jakarta Sans): the same pair the site self-hosts.
%
% Accessibility: WCAG 2.1 AA. Contrast measured against the white page —
%   ink   #101113  18.89:1  body text   (the site's ground colour, inverted)
%   blue  #0072B2   5.19:1  headings, links   (Okabe-Ito, colour-blind safe)
%   slate #646E7C   5.17:1  secondary text
% All three clear 4.5:1, so each is safe at NORMAL text size, not merely large.
% Colour is never the sole cue: headings also carry weight, size, and a rule,
% and every link's URL is spelled out as visible text.
%
% Deliberately NOT carried over from the site spectrum — each fails AA for
% normal text on a light ground:
%   amber #E69F00 2.25:1 · pink #CC79A7 3.06:1 · teal #009E73 3.42:1
%   brick #E05A4A 3.66:1 · indigo #7A79C0 3.94:1 · siue-red #E5182D 4.67:1
% Several would pass for large text only. The CV needs no second accent, and a
% single accent keeps the light rendering calm.

% ── Colors ──────────────────────────────────────────────────────────────────
\usepackage{xcolor}
\definecolor{instink}{HTML}{101113}    % body text
\definecolor{instblue}{HTML}{0072B2}   % accent: headings, links
\definecolor{instslate}{HTML}{646E7C}  % secondary text
\definecolor{instrule}{HTML}{C9CDD3}   % hairlines (decorative, non-text)

% ── Fonts ───────────────────────────────────────────────────────────────────
\usepackage{fontspec}
\defaultfontfeatures{Ligatures=TeX}

\setmainfont{Jakarta}[
  Path = ./fonts/, Extension = .ttf,
  UprightFont = *-Regular, BoldFont = *-Bold,
  ItalicFont = *-Italic, BoldItalicFont = *-BoldItalic,
]
\setsansfont{Archivo}[
  Path = ./fonts/, Extension = .ttf,
  UprightFont = *-Regular, BoldFont = *-Bold,
  ItalicFont = *-Italic, BoldItalicFont = *-BoldItalic,
]

% ── Even out justification ──────────────────────────────────────────────────
% pandoc's template already loads microtype, so options go through
% PassOptionsToPackage — loading it again here is an option clash. Protrusion
% pulls punctuation slightly into the margin; expansion stays off because
% xelatex does not support it.
\PassOptionsToPackage{protrusion=true, expansion=false}{microtype}

% Let the line breaker work a little harder before it opens up word gaps.
% emergencystretch gives it a last-resort slack line rather than a river.
\tolerance=1400
\emergencystretch=1.2em
\hyphenpenalty=120

% ── Section styling: lift \subsection to act as the top-level section ───────
\usepackage{titlesec}
\titleformat{\subsection}
  {\Large\bfseries\sffamily\color{instblue}}{}{0pt}
  {}[\vspace{-3pt}\color{instrule}\titlerule]
\titlespacing*{\subsection}{0pt}{18pt}{8pt}

\titleformat{\subsubsection}
  {\normalsize\bfseries\sffamily\color{instslate}}{}{0pt}{}
\titlespacing*{\subsubsection}{0pt}{11pt}{3pt}

% ── Paragraph + list rhythm ─────────────────────────────────────────────────
\setlength{\parindent}{0pt}
\setlength{\parskip}{5.5pt plus 1.5pt minus 0.5pt}

\usepackage{enumitem}
\setlist[itemize]{itemsep=2.5pt, topsep=4pt, parsep=0pt, leftmargin=15pt,
                  label=\textcolor{instblue}{\textbullet}}

% ── Footer: name + page number, no header line ──────────────────────────────
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyfoot[L]{\footnotesize\color{instslate}Alex P. Leith, PhD --- CV}
\fancyfoot[R]{\footnotesize\color{instslate}\thepage}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}

\fancypagestyle{plain}{%
  \fancyhf{}
  \fancyfoot[L]{\footnotesize\color{instslate}Alex P. Leith, PhD --- CV}
  \fancyfoot[R]{\footnotesize\color{instslate}\thepage}
  \renewcommand{\headrulewidth}{0pt}
  \renewcommand{\footrulewidth}{0pt}
}

% ── Body colour ─────────────────────────────────────────────────────────────
\AtBeginDocument{\color{instink}}

% ── Links + PDF metadata ────────────────────────────────────────────────────
% Deferred because pandoc loads hyperref after this header. pdflang and
% pdftitle let assistive technology announce the document correctly; the
% pandoc/xelatex path emits no full tag tree, so this is the accessible
% ceiling for this pipeline (see repo README).
\AtBeginDocument{%
  \hypersetup{
    colorlinks=true,
    linkcolor=instblue,
    urlcolor=instblue,
    citecolor=instblue,
    pdfborder={0 0 0},
    pdftitle={Alex P. Leith, PhD --- Curriculum Vitae},
    pdfauthor={Alex P. Leith},
    pdfsubject={Curriculum Vitae},
    pdflang={en-US},
    pdfdisplaydoctitle=true
  }%
  \urlstyle{same}%
}

% ── Penalize widows + orphans aggressively for cleaner pagination ───────────
\widowpenalty=10000
\clubpenalty=10000

% ── Inter-paragraph rhythm for entry-style content ──────────────────────────
\linespread{1.12}

% ── Tighten title block spacing ─────────────────────────────────────────────
% Default pandoc \maketitle leaves an enormous gap between the subtitle line
% and the first body paragraph because the empty date block + post-title
% vspace add up to ~80-100pt of dead air. Pull the title block up toward
% the top margin and collapse the empty-date region with a stronger
% negative vspace so the body lands within ~16pt of the subtitle.
\usepackage{titling}
\setlength{\droptitle}{-40pt}
\pretitle{\begin{center}\sffamily\bfseries\Huge}
\posttitle{\par\end{center}\vspace{-4pt}}
\predate{\par\begin{center}\sffamily\color{instslate}}
\postdate{\par\end{center}\vspace{-44pt}}
